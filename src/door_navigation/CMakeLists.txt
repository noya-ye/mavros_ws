cmake_minimum_required(VERSION 3.16)
project(door_navigation VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(nlohmann_json REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_library(door_core STATIC src/core.cpp src/cloud.cpp)
target_include_directories(door_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(door_core PUBLIC Eigen3::Eigen ${OpenCV_LIBS})
add_executable(door_node src/door_node.cpp)
target_link_libraries(door_node PRIVATE door_core nlohmann_json::nlohmann_json)
ament_target_dependencies(door_node PUBLIC rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs trajectory_msgs)
install(TARGETS door_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
