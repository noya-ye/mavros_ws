cmake_minimum_required(VERSION 3.16)
project(door_offboard LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(nlohmann_json REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/PlannerEnable.msg" "msg/DoorGoal.msg" "msg/GoalReached.msg"
  DEPENDENCIES std_msgs geometry_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_library(door_core STATIC src/core.cpp src/cloud.cpp)
target_include_directories(door_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(door_core PUBLIC Eigen3::Eigen ${OpenCV_LIBS})
foreach(node perception_node flight_node)
  add_executable(${node} src/${node}.cpp)
  target_link_libraries(${node} PRIVATE door_core nlohmann_json::nlohmann_json "${cpp_typesupport_target}")
  ament_target_dependencies(${node} PUBLIC rclcpp std_msgs geometry_msgs sensor_msgs nav_msgs mavros_msgs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
